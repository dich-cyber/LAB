l=[]
n = int(input())
for i in range(n):
    st = list(map(int, input().split()))
    l.append(st)
p=[]
m = int(input())
for i in range(m):
    st = list(map(int, input().split()))
    p.append(st)

k = []
for z in range(n):
    row = []

    for x in range(m):
        s = 0

        for c in range(m):
            s = s + l[z][c] * p[x][c]

        row.append(s)

    k.append(row)
print(k)
